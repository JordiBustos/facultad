\section{Introducción}

Sea \( \mu \) una medida de Borel en \( \R^k \) y \( m \) la medida de Lebesgue en \( \R^k \).
Nuestro objetivo es definir la derivada de \( \mu \) en un punto \( x \in \R^k \) como el límite del cociente:
\[
    \frac{\mu(I)}{m(I)}
\]
cuando \( I \) es un cubo que contiene a \( x \) y su diámetro tiende a cero. Esta formalización se puede hacer con bolas,
recordemos que dado \( x \in \R^k \) y \( r > 0 \), definimos la bola abierta como: \[
    B(x, r) = \{ y \in \R^k : \| y - x \| < r \}
\]
Asociada a la medida \( \mu \) consideramos el cociente entre su medida y la medida de Lebesgue en la bola: \[
    Q_r \mu (x) = \frac{\mu(B(x, r))}{m(B(x, r))}
\]
Definimos la derivada simétrica de \( \mu \) en \( x \) como el límite de este cociente cuando el radio tiende a cero: \[
    D \mu (x) = \lim_{r \to 0} Q_r \mu (x)
\]
en los puntos donde el límite existe.

Para estudiar la existencia de este límite y las propiedades de la función \( D \mu \) es fundamental introducior la \textbf{función maximal} \( M \mu \), definida por: \[
    M \mu (x) = \sup_{0 < r < \infty} Q_r \mu (x)
\]

\begin{teorema}[Descomposición de Hahn]
    Si \( \lambda \) es una medida con signo en \( \mathfrak{X} \) entonces existen dos conjuntos \( P, N \in \mathfrak{X} \) con \( X = P \cup N \) y
    \( P \cap N = \varnothing \) tal que \( \lambda(E \cap P) \geq 0 \) y \( \lambda(E \cap N) \leq 0 \) para cada \( E \in \mathfrak{X} \).
\end{teorema}

\begin{definicion}[Variación total]
    Sea \( \lambda \) una medida con signo o carga en \( \R^k \). Sean \( P, N \) la descomposición de Hahn de \( \lambda \). La descomposición positiva y negativa de \( \lambda \) son las medidas finitas \( \lambda^+ \) y \( \lambda^- \) respectivamente, definidas por \( \lambda^+(E) = \lambda(E \cap P) \) y \( \lambda^-(E) = -\lambda(E \cap N) \) para cada boreliano \( E \). La variación total de \( \lambda \) es la medida finita \( |\lambda| = \lambda^+ + \lambda^- \).
\end{definicion}

Si \( \mu \) es una medida con signo o carga tomamos: \[
    M \mu(x) = \sup_{0 < r < \infty} \frac{|\mu|(B(x, r))}{m(B(x, r))}
\]
donde \( |\mu| \) es la de antes.

\begin{definicion}
    Dada una función \( f: \R^k \to \R \cup \{ \infty \} \), se dice que \( f \) es \textbf{semi-continua inferiormente} si para cada \( t \in \R \), el conjunto \( \{ x : f(x) > t \} \) es abierto.
\end{definicion}

\begin{proposicion}
    \( M \mu \) es semi-continua inferiormente y, por lo tanto, medible.
\end{proposicion}
\begin{proof}
    Para ello, consideremos \( \mu \geq 0 \), \( \lambda > 0 \) y \( E = \{ M \mu(x) > \lambda \} \). Basta ver que \( E \) es abierto, veamos que cada \( x \in E \) es interior.

    En efecto, existe \( r > 0 \) tal que \[
        \mu(B(x,r)) = t m(B(x,r))
    \]
    para algún \( t > \lambda \). Además, existe \( \delta > 0 \) tal que:\[
        {(r+\delta)}^k < \frac{t}{\lambda} r^k \implies \lambda < \frac{t \cdot r^k}{{(r+\delta)}^k} \implies t {\left(\frac{r}{r+\delta}\right)}^k > \lambda
    \]
    Pues \( \frac{t}{\lambda} > 1 \implies r^k < \frac{t}{\lambda} \cdot r^k \) y, por continuidad de la función \( r \mapsto r^k \), existe \( \delta > 0 \) tal que \( {(r+\delta)}^k < \frac{t}{\lambda} r^k \).

    Si \( \| y - x \| < \delta \implies B(x,r) \subset B(y, r+\delta) \) y, por lo tanto, \begin{align*}
        \mu(B(y, r+\delta)) \geq \mu(B(x, r)) & = t m(B(x,r))                                               \\
                                              & = t {\left( \frac{r}{r+\delta} \right)}^k m(B(y, r+\delta)) \\
                                              & > \lambda m(B(y, r+\delta))
    \end{align*}
    Esto implica que \( y \in E \), notar que para la última igualdad utilizamos que:\begin{align*}
        m(B(x, r))        & = V_k r^k            \\
        m(B(y, r+\delta)) & = V_k {(r+\delta)}^k
    \end{align*}
    donde \( V_k \) es el volumen de la bola unitaria en \( \R^k \).
    Luego, despejando \( V_k = \frac{m(B(y, r+\delta))}{{(r+\delta)}^k} \) y sustituyendo en la expresión de \( m(B(x, r)) \) obtenemos que:\[
        m(B(x, r)) = \frac{m(B(y, r+\delta))}{{(r+\delta)}^k} \cdot r^k = {\left( \frac{r}{r+\delta} \right)}^k m(B(y, r+\delta))
    \]
    Finalmente, \( B(x, \delta) \subset E \) y \( E \) es abierto \( \therefore M \mu \) es medible.

    Si \( \mu \) es una carga entonces vale pues \( |\mu| \geq 0 \) y \( M \mu = M |\mu| \).
\end{proof}

Probaremos el siguiente lemma que nos ayudará a demostrar el Teorema Maximal.
\begin{lemma}[Lema de cubrimiento finito de Vitali]
    Sea \( W \) la unión finita de bolas abiertas de la forma \( B(x_i, r_i) \) para \( i = 1, \dots, n \). Entonces, existe un subconjunto \( S \subseteq \{ 1, \dots, n \} \) tal que las bolas \( B(x_i, r_i) \) para \( i \in S \) son disjuntas dos a dos, \( W \subseteq \bigcup_{i \in S} B(x_i, 3r_i) \) y \( m(W) \leq 3^k \sum_{i \in S} m(B(x_i, r_i)) \).
\end{lemma}

\begin{proof}
    Ordenemos las bolas de tal forma que \( r_1 \geq r_2 \geq \cdots \geq r_n \). Construiremos \( S \) de la siguiente manera: \( 1 \in S \) y descartamos todas las bolas \( B(x_j, r_j) \) que intersectan a \( B(x_1, r_1) \).

    Luego, sea \( i_2 \) el menor índice tal que \( B(x_{i_2}, r_{i_2}) \) no intersecta a \( B(x_1, r_1) \). Agregamos \( i_2 \) a \( S \) y descartamos todas las bolas \( B(x_j, r_j) \) que intersectan a \( B(x_{i_2}, r_{i_2}) \). Repetimos este proceso hasta que no queden bolas. Por construcción, es claro que las bolas \( B(x_i, r_i) \) para \( i \in S \) son disjuntas.

    Sea \( z \in W \), entonces \( z \in B(x_i, r_i) \) para algún \( i \). Si \( i \in S \), entonces trivialmente \( z \in B(x_i, 3r_i) \). Si \( i \notin S \), entonces la bola \( B(x_i, r_i) \) fue descartada porque intersecta a \( B(x_j, r_j) \) para algún \( j \in S \) con \( r_j \geq r_i \). Luego, existe \( y \in B(x_i, r_i) \cap B(x_j, r_j) \). Por lo tanto, utilizando la desigualdad triangular:
    \begin{align*}
        d(z, x_j) & \leq d(z, x_i) + d(x_i, x_j)          \\
                  & < d(z, x_i) + [d(x_i, y) + d(y, x_j)] \\
                  & < d(z, x_i) + 2r_j                    \\
                  & < r_i + 2r_j                          \\
                  & \leq 3r_j
    \end{align*}
    Luego, \( z \in B(x_j, 3r_j) \) y, en consecuencia, \( W \subseteq \bigcup_{i \in S} B(x_i, 3r_i) \).

    Además, utilizando la subaditividad de la medida de Lebesgue y su comportamiento frente a dilataciones, obtenemos:
    \[
        m(W) \leq m\left( \bigcup_{i \in S} B(x_i, 3r_i) \right) \leq \sum_{i \in S} m(B(x_i, 3r_i)) = 3^k \sum_{i \in S} m(B(x_i, r_i))
    \]
    lo que concluye la demostración.
\end{proof}

El siguiente teorema nos dice que la función maximal no puede ser grande en un conjunto grande.

\begin{teorema}[Teorema Maximal]
    Si \( \mu \) es una medida de Borel con signo o carga en \( \R^k \) y \( \lambda > 0 \), entonces:
    \[
        m\left(\{ x \in \R^k \, : \, (M\mu)(x) > \lambda \}\right) \leq \frac{3^k}{\lambda} |\mu|(\R^k)
    \]
    donde \( |\mu| \) es la variación total de \( \mu \).
\end{teorema}
\begin{proof}
    Fijado \( \mu \) y \( \lambda > 0 \), consideremos \( K \) un conjunto compacto contenido en el abierto \( \{ M \mu > \lambda \} \).
    Para cada \( x \in K \), existe \( r_x > 0 \) tal que \( Q_{r_x} \mu(x) > \lambda \). Por lo tanto, \( K \subseteq \bigcup_{x \in K} B(x, r_x) \). Como \( K \) es compacto, existe una subcolección finita de bolas \( {\{ B(x_i, r_i) \}}_{i = 1}^n \) que cubre a \( K \).

    Aplicando el lema anterior, existe un subconjunto \( S \subseteq \{ 1, \ldots, n \} \) tal que las bolas \( B(x_i, r_i) \) para \( i \in S \) son dos a dos disjuntas y cumplen las propiedades de contención del lema. Luego:
    \[
        m(K) \leq 3^k \sum_{i \in S} m(B(x_i, r_i)) < \frac{3^k}{\lambda} \sum_{i \in S} |\mu|(B(x_i, r_i)) \leq \frac{3^k}{\lambda} |\mu|(\R^k)
    \]

    Recordemos que con la medida de Lebesgue \( m \) podemos aproximar la medida de un conjunto abierto a través de de compactos contenidos en él i.e el supremo de las medidas de los compactos contenidos en el conjunto abierto nos da la medida del conjunto:
    \[
        m(\{ M \mu > \lambda \}) = \sup \{ m(K) : K \subseteq \{ M \mu > \lambda \}, K \text{ compacto} \} \leq \frac{3^k}{\lambda} |\mu|(\R^k)
    \]
    lo que concluye la demostración.
\end{proof}

\begin{definicion}[Espacio \( L_1 \) débil]
    Se dice que una función medible \( f \) en \( \R^k \) pertenece al espacio \( L_1 \) débil si existe una constante \( C > 0 \) tal que, para todo \( \lambda > 0 \):
    \[
        m \left( \{ x \in \R^k : |f(x)| > \lambda \} \right) \leq \frac{C}{\lambda}
    \]
    (Es decir, la función \( \lambda \mapsto \lambda \cdot m \{ |f| > \lambda \} \) está acotada en \( (0, \infty) \)).
\end{definicion}

\begin{observacion}
    Toda función \( f \in L_1(\R^k) \) pertenece a \( L_1 \) débil. En efecto, si tomamos \( E = \{ |f| > \lambda \} \), se cumple que:
    \[
        \lambda m(E) \leq \int_E |f| \, \mathrm{d} m \leq \int_{\R^k} |f| \, \mathrm{d} m = \|f\|_1
    \]
    Por lo tanto, \( m(E) \leq \frac{\|f\|_1}{\lambda} \), satisfaciendo la definición con \( C = \|f\|_1 \).
\end{observacion}

Para cada \( f \in L^1(\R^k) \) asociamos la función maximal \( Mf: \R^k \to [0, \infty] \) definida por:\[
    Mf(x) = \sup_{0 < r < \infty} \frac{1}{m(B(x, r))} \int_{B(x, r)} |f| \, \mathrm{d} m
\]

Notemos que si pensamos a \( M \) como un operador, este mapea funciones de \( L_1 \) a funciones de \( L_1 \) débil, es decir, \( Mf \) puede no ser integrable.

Si identificamos \( f \) con la medida \( \mu \) dada por \( d \mu = f d m \), entonces \( Mf = M \mu \). Por lo tanto, el Teorema Maximal implica que \( Mf \) es una función
de \( L_1 \) débil, es decir, existe una constante \( C > 0 \) tal que para todo \( \lambda > 0 \):
\[
    m \left( \{ x \in \R^k : Mf(x) > \lambda \} \right) \leq \frac{C}{\lambda}
\]
tomando \( C = 3^k \|f\|_1 \) se cumple esta desigualdad, lo que muestra que \( Mf \) es de \( L_1 \) débil.

\begin{definicion}[Punto de Lebesgue]
    Dado \( f \in L_1(\R^k) \), se dice que \( x \in \R^k \) es un punto de Lebesgue de \( f \) si cumple que: \[
        \lim_{r \to 0} \frac{1}{m(B(x, r))} \int_{B(x, r)} |f(y) - f(x)| \, \mathrm{d} m(y) = 0
    \]
\end{definicion}

Por ejemplo, para una función continua todo punto del dominio es un punto de Lebesgue. En general, que un punto \( x \in \R^k \) sea de Lebesgue nos dice que los promedios
\( |f - f(x)| \) son pequeños en bolas pequeñas alrededor de \( x \).

Demostraremos un resultado muy útil respecto a la medida de Lebesgue que nos ayudara en la demostración del siguiente teorema.

\begin{definicion}
    Un subconjunto \( E \subseteq \R^k \) con \( m^*(E) = 0 \) se llama un \textbf{conjunto de Lebesgue nulo}.
\end{definicion}

\begin{proposicion}\label{proposicion:conjuntos-nulos}
    Si \( E \subseteq \R^k \) es un conjunto de Lebesgue nulo, entonces \( E \) es medible Lebesgue y \( m(E) = 0 \). Es más, todo subconjunto de \( E \) es medible y tiene medida cero.
\end{proposicion}
\begin{proof}
    Recordemos que un conjunto \( E \subseteq \R^k \) satisface la condición de Carathéodory \( \iff \) para cada \( A \subseteq \R^k \) tal que \( m^*(A) < \infty \), se cumple que \[
        m^*(A) \geq m^*(A \cap E) + m^*(A \setminus E)
    \]
    En efecto, fijado \( E \subseteq \R^k \) tal que \( m^*(E) = 0 \) y sea \( A \subseteq \R^k \) con \( m^*(A) < \infty \),
    como \( A \cap E \subseteq E \) y \( A \setminus E \subseteq A \), se cumple por la monotonía de \( m^* \) que \[
        m^*(A) = m^*(E) + m^*(A) \geq m^*(A \cap E) + m^*(A \setminus E)
    \]
    Luego, \( E \) es medible y \( m(E) = m^*(E) = 0 \).

    Si \( B \subseteq E \), entonces \( m^*(B) \leq m^*(E) = 0 \implies m^*(B) = 0 \therefore B \) es medible y \( m(B) = 0 \).
\end{proof}

El siguiente teorema nos dice que casi todo punto es un punto de Lebesgue para funciones en \( L_1 \).

\begin{teorema}
    Si \( f \in L_1(\mathbb{R}^k) \), entonces casi todo punto \( x \in \mathbb{R}^k \) es un punto de Lebesgue de \( f \).
\end{teorema}
\begin{proof}
    Consideremos la función de los promedios para \( x \in \mathbb{R}^k \) y \( r > 0 \):
    \[
        (T_r f)(x) = \frac{1}{m(B(x,r))} \int_{B(x,r)} |f(y) - f(x)| \, \mathrm{d}m(y)
    \]
    Como a priori no conocemos si el límite ordinario cuando \( r \to 0 \) existe,
    definimos el operador utilizando el límite superior, el cual está siempre bien definido en la recta real extendida:
    \[
        Tf(x) = \limsup_{r \to 0} \, (T_r f)(x)
    \]
    Nuestro objetivo es probar que \(Tf(x) = 0\) para casi todo punto.
    Dado que \((T_r f)(x) \ge 0\) para todo \(r\), demostrar que su límite superior es menor o igual a cero implicará,
    por el Teorema del Sándwich, que el límite ordinario existe y vale cero.

    Fijemos \(\varepsilon > 0\) y sea \(n \in \mathbb{N}\).
    Dado que el espacio de las funciones continuas con soporte compacto \(C_c(\mathbb{R}^k)\) es denso en \(L_1(\mathbb{R}^k)\),
    existe una función \(g \in C_c(\mathbb{R}^k)\) tal que \( \|f - g\|_1 < \frac{1}{n}\).

    \begin{observacion}
        Es crucial exigir que \(g\) tenga soporte compacto y no simplemente que sea continua. Como \(f \in L_1(\mathbb{R}^k)\), para que la diferencia \(h = f - g\) pertenezca a \(L_1(\mathbb{R}^k)\) (requisito indispensable para aplicar el Teorema Maximal), se necesita obligatoriamente que \(g \in L_1(\mathbb{R}^k)\). Una función continua arbitraria en \(\mathbb{R}^k\) podría tener integral infinita, mientras que el soporte compacto garantiza la finitud de su integral.
    \end{observacion}

    Definamos \(h = f - g\), de modo que \(f = g + h\). Por la desigualdad triangular en el integrando de los promedios, es inmediato que:
    \[
        |f(y) - f(x)| \le |g(y) - g(x)| + |h(y) - h(x)|
    \]
    Integrando sobre la bola \(B(x,r)\) y dividiendo por su medida, obtenemos la subaditividad para cada promediación fija:
    \[
        (T_r f)(x) \le (T_r g)(x) + (T_r h)(x)
    \]
    Aplicando la propiedad fundamental de subaditividad del límite superior (\(\limsup(A+B) \le \limsup A + \limsup B\)) al tomar \(r \to 0\), deducimos:
    \[
        Tf(x) \le Tg(x) + Th(x)
    \]
    Como \(g\) es una función continua, vale que \(Tg(x) = 0\) pues \( x \) es punto de Lebesgue para \( g \) (\(\forall x \in \R^k \)). Por lo tanto:
    \[
        Tf(x) \le Th(x)
    \]
    Por otro lado, podemos acotar el operador \((T_r h)(x)\) utilizando la función maximal \(Mh(x)\):
    \[
        (T_r h)(x) \le \frac{1}{m(B(x,r))} \int_{B(x,r)} |h(y)| \, \mathrm{d}m(y) + |h(x)| \le Mh(x) + |h(x)|
    \]
    Tomando el límite superior cuando \(r \to 0\) en dicha desigualdad, resulta que \(Tf(x) \le Th(x) \le Mh(x) + |h(x)|\).

    En consecuencia, si para un punto \(x\) se cumple que \(Tf(x) > 2\varepsilon \), necesariamente debe ocurrir que \(Mh(x) > \varepsilon \) o bien \(|h(x)| > \varepsilon \). Esto nos permite realizar la siguiente inclusión de conjuntos:
    \[
        \{x \in \mathbb{R}^k : Tf(x) > 2\varepsilon \} \subseteq \{x \in \mathbb{R}^k : Mh(x) > \varepsilon \} \cup \{x \in \mathbb{R}^k : |h(x)| > \varepsilon \}
    \]
    Tomando la medida exterior de Lebesgue \(m^*\) en ambos miembros, y aplicando la subaditividad de la misma junto con el Teorema Maximal para \( Mh \in L_1 \) débil y la desigualdad de Chebyshev para \(|h|\), obtenemos:
    \begin{align*}
        m^*(\{Tf > 2\varepsilon \}) & \le m(\{Mh > \varepsilon \}) + m(\{|h| > \varepsilon \})            \\
                                    & \le \frac{3^k}{\varepsilon} \|h\|_1 + \frac{1}{\varepsilon} \|h\|_1 \\
                                    & = \frac{3^k + 1}{\varepsilon} \|h\|_1                               \\
                                    & < \frac{3^k + 1}{\varepsilon \cdot n}
    \end{align*}
    Notemos que el conjunto del miembro izquierdo, \( \{Tf > 2\varepsilon \} \), es un conjunto fijo e independiente de la elección de \(n\). Como la cadena de desigualdades es válida para cualquier \(n \in \mathbb{N}\) arbitrario, podemos tomar el límite cuando \(n \to \infty \), concluyendo con seguridad que:
    \[
        m^*(\{Tf > 2\varepsilon \}) = 0
    \]
    Por la arbitrariedad de \(\varepsilon > 0\), si definimos el conjunto de los puntos que no son de Lebesgue como \(E = \{Tf > 0\} \), se observa que \(E = \bigcup_{j=1}^{\infty} \{Tf > \frac{1}{j}\} \). Por la subaditividad numerable de la medida exterior, obtenemos que \(m^*(E) = 0\).

    Finalmente, por la \textbf{Proposición\( \, \)\ref{proposicion:conjuntos-nulos}} de nuestro desarrollo (completitud de la medida de Lebesgue), al ser \(E\) un subconjunto con medida exterior nula, concluimos que \(E\) es un conjunto Lebesgue medible y su medida es efectivamente \(m(E) = 0\). Esto demuestra que \(Tf(x) = 0\) para casi todo punto, evitando la necesidad de demostrar previamente la medibilidad de la función límite \(Tf\).
\end{proof}

\begin{observacion}\label{obs:promedios-lebesgue}
    Si \( x \in \R^k \) es un punto de Lebesgue de \( f \in L_1(\R^k) \), entonces:\[
        \lim_{r \to 0} \frac{1}{m(B(x, r))} \int_{B(x, r)} |f(y) - f(x)| \, \mathrm{d} m(y) = 0 \\
    \]
    Si acotamos la diferencia entre:\begin{align*}
        \left | \left( \frac{1}{m(B(x, r))} \int_{B(x, r)} f(y) \, \mathrm{d} m(y) \right) - f(x) \right | & = \left | \frac{1}{m(B(x, r))} \int_{B(x, r)} (f(y) - f(x)) \, \mathrm{d} m(y) \right | \\
                                                                                                           & \leq \frac{1}{m(B(x, r))} \int_{B(x, r)} |f(y) - f(x)| \, \mathrm{d} m(y)
    \end{align*}
    El lado derecho de la desigualdad tiende a \( 0 \) si \( r \to 0 \) pues \( x \) es un punto de Lebesgue de \( f \). Esto implica que:\[
        \lim_{r \to 0} \frac{1}{m(B(x, r))} \int_{B(x, r)} f(y) \, \mathrm{d} m(y) = f(x)
    \]
\end{observacion}

\begin{teorema}\label{thm:radon-nikodym}
    Sea \( \mu \) una medida de Borel con signo o carga en \( \R^k \) tal que \( \mu \ll m \). Sea \( f \) la derivada de Radon-Nikodym de \( \mu \) con respecto a \( m \).
    Entonces \( D \mu(x) = f(x) \) \( m \)-c.t.p y \[
        \mu(E) = \int_E D \mu \, \mathrm{d} m
    \]
    para todo \( E \) Boreliano de \( \R^k \).

    Es decir, la derivada de Radon-Nikodym se puede obtener como el límite de los cocientes \( Q_r \mu(x) \) cuando \( r \to 0 \) para casi todo punto \( x \in \R^k \).
\end{teorema}
\begin{proof}
    Como \( \mu \) es carga, entonces \( |\mu|(\R^k) < \infty \) y el teorema de Radon-Nikodym nos asegura que existe \( f \in L_1(\R^k) \) tal que \[
        \mu(E) = \int_E f \, \mathrm{d} m
    \]
    para todo \( E \) Boreliano de \( \R^k \). Sea \( x \in \R^k \) un punto de Lebesgue de \( f \), entonces:\[
        f(x) = \lim_{r \to 0} \frac{1}{m(B(x, r))} \int_{B(x, r)} f \, \mathrm{d} m = \lim_{r \to 0} \frac{\mu(B(x, r))}{m(B(x, r))} = D \mu(x)
    \]
    Como vimos que para una función \( f \in L_1(\R^k) \) casi todo punto es un punto de Lebesgue, se concluye que \( D \mu(x) = f(x) \) para casi todo \( x \in \R^k \). Por lo tanto, para cada \( E \) Boreliano de \( \R^k \):\[
        \int_E D \mu \, \mathrm{d} m = \int_E f \, \mathrm{d} m = \mu(E)
    \]
\end{proof}

\section{Diferenciación sobre familias que se contraen}

\begin{definicion}[Conjuntos que se contraen adecuadamente]
    Dado \( x \in \R^k \). Una familia de conjuntos de Borel en \( \R^k \) \( {\{ E_i \}}_{i \in \N} \) se dice que se \textbf{contrae adecuadamente} a \( x \)
    si existe \( \alpha > 0 \) tal que existe una sucesión de bolas \( B(x, r_i) \) con \( r_i \to 0 \) tal que \( E_i \subseteq B(x, r_i) \) y \[
        m(E_i) \geq \alpha \cdot m(B(x, r_i)) \quad \forall i \in \N
    \]
\end{definicion}

Notemos que en la defición no se requiere que \( x \in E_i \) o, ni siquiera, que \( x \in \overline{E_i} \) para cada \( i \in \N \).

Veamos un ejemplo de conjuntos que se contraen adecuadamente a un punto \( x \in \R^2 \) y que no son bolas.
Consideremos los cuadrados de \( \R^2 \) inscritos en las bolas \( B(x, r) \) con \( r \to 0 \). Estos conjuntos se contraen adecuadamente a \( x \) pues se puede ver que cada cuadrado tiene lado \( r \sqrt{2} \) y, por lo tanto,
la medida de Lebesgue de cada cuadrado es \( 2 r^2 \) mientras que la medida de Lebesgue de la bola es \( \pi r^2 \). Tomando \( \alpha = \frac{2}{\pi} \) se cumple la condición de la definición.

\begin{teorema}[Teorema de Diferenciación de Lebesgue]
    Asociemos a cada \( x \in \R^k \) una sucesión de conjuntos de Borel \( {\{ E_i(x) \}}_{i \in \N} \) que se contrae adecuadamente a \( x \). Si \( f \in L_1(\R^k) \), entonces:
    \[
        f(x) = \lim_{i \to \infty} \frac{1}{m(E_i(x))} \int_{E_i(x)} f \, \mathrm{d} m
    \]
    para todo \( x \in \R^k \) que sea un punto de Lebesgue de \( f \).
\end{teorema}

\begin{proof}
    Sea \( x \in \R^k \) un punto de Lebesgue de \( f \). Sean \( \alpha(x) > 0 \) y \( {\{ B(x, r_i) \}}_{i \in \N} \) la constante y la sucesión de bolas que garantizan que \( {\{ E_i(x) \}}_{i \in \N} \) se contrae adecuadamente a \( x \). Entonces, para cada \( i \in \N \):
    \[
        \frac{\alpha(x)}{m(E_i(x))} \int_{E_i(x)} |f - f(x)| \, \mathrm{d} m \leq \frac{1}{m(B(x, r_i))} \int_{B(x, r_i)} |f - f(x)| \, \mathrm{d} m
    \]
    Como \( x \) es un punto de Lebesgue de \( f \), el lado derecho de la desigualdad converge a \( 0 \) cuando \( i \to \infty \). Por lo tanto, el lado izquierdo también converge a \( 0 \) y se tiene que:
    \[
        \lim_{i \to \infty} \frac{1}{m(E_i(x))} \int_{E_i(x)} |f - f(x)| \, \mathrm{d} m = 0
    \]
    Luego, análogamente a la Observación\( \, \)\ref{obs:promedios-lebesgue} que hicimos se obtiene:
    \[
        \lim_{i \to \infty} \frac{1}{m(E_i(x))} \int_{E_i(x)} f \, \mathrm{d} m = f(x) \quad m\text{-c.t.p.}
    \]
    lo que concluye la demostración.

    Cabe destacar que en la demostración no se está pidiendo ninguna relación entre los conjuntos que se contraen adecuadamente a puntos diferentes, es decir, cada punto \( x \in \R^k \) puede tener su propia familia de conjuntos de Borel que se contrae adecuadamente a \( x \).
\end{proof}

\section{Descomposición de Lebesgue y diferenciación de medidas}

El teorema anterior nos da una versión más fuerte del Teorema\( \, \)\ref{thm:radon-nikodym}. Veamos los preliminares necesarios para la exposición del resultado final.

\begin{definicion}
    Decimos que dos medidas \( \mu, \lambda \) en \( \X \) son mutuamente singulares si existen \( A, B \in X \) disjuntos tales que \( A \cup B = X \) y \( \lambda(A) = \mu(B) = 0 \).
    Lo notamos como \( \lambda \perp \mu \).
\end{definicion}

\begin{teorema}[Teorema de la descomposición de Lebesgue]
    Sea \( \mu \) y \( \lambda \) dos medidas \( \sigma \)-finitas definidas sobre una \( \sigma \)-álgebra \( \X \). Entonces, existen \( \lambda_1, \lambda_2 \) medidas tales que \( \lambda_1 \perp \mu \) y
    \( \lambda_2 \ll \mu \) y \( \lambda = \lambda_1 + \lambda_2 \).
\end{teorema}
\begin{proof}
    Sea \( \nu = \lambda + \mu \). Como \( \lambda \ll \nu \) y \( \mu \ll \nu \), por Radon-Nikodym existen funciones \( f,g \in L_1(\nu) \) tales que
    \[
        \lambda(E)=\int_E f\,\mathrm d\nu
        \qquad\text{y}\qquad
        \mu(E)=\int_E g\,\mathrm d\nu
    \]
    para todo \( E\in\X \).

    Definimos
    \[
        A=\{g=0\}, \qquad B=\{g>0\},
    \]
    y, para cada \( E\in\X \),
    \[
        \lambda_1(E)=\lambda(E\cap A),
        \qquad
        \lambda_2(E)=\lambda(E\cap B).
    \]
    Entonces \( A\cap B=\varnothing \), \( A\cup B=\X \), y por lo tanto
    \[
        \lambda(E)=\lambda(E\cap A)+\lambda(E\cap B)=\lambda_1(E)+\lambda_2(E).
    \]

    Además,
    \[
        \mu(A)=\int_A g\,\mathrm d\nu=0,
        \qquad
        \lambda_1(B)=\lambda(B\cap A)=0,
    \]
    de modo que \( \lambda_1\perp\mu \).

    Falta probar \( \lambda_2\ll\mu \). Sea \( E\in\X \) con \( \mu(E)=0 \). Entonces
    \[
        0=\mu(E)=\int_E g\,\mathrm d\nu.
    \]
    Como \( g\ge 0 \), se sigue que \( g=0 \) \( \nu \)-c.t.p. en \( E \), luego \( \nu(E\cap B)=0 \). Dado que \( \lambda\ll\nu \), obtenemos
    \[
        \lambda_2(E)=\lambda(E\cap B)=0.
    \]
    Por consiguiente, \( \lambda_2\ll\mu \), y queda demostrada la descomposición.
\end{proof}

\begin{teorema}[Diferenciación de Medidas de Borel]
    Sea \( \mu \) una medida de Borel con signo o carga en \( \R^k \) y sea \( \mu = \mu_a + \mu_s \) su descomposición de Lebesgue respecto a la medida de Lebesgue \( m \) (con \( \mu_a \ll m \) y \( \mu_s \perp m \)).

    Sea \( f \in L_1(\R^k) \) la derivada de Radon-Nikodym de \( \mu_a \) (es decir, \( \mathrm{d}\mu_a = f \, \mathrm{d}m \)). Entonces, la derivada simétrica \( D\mu(x) \) existe \( m \)-casi en todo punto y satisface:
    \[
        D\mu(x) = f(x) \quad m\text{-c.t.p.}
    \]

    Como consecuencia directa, si \( \mu \ll m \) (luego \( \mu_s = 0 \)), entonces \( D\mu = f \) \( m \)-c.t.p y obtenemos la fórmula explícita:
    \[
        \mu(E) = \int_E (D\mu) \, \mathrm{d}m
    \]
    para todo conjunto de Borel \( E \).
\end{teorema}
\begin{proof}
    Descomponemos \( \mu=\mu_a+\mu_s \), donde \( \mu_a\ll m \) y \( \mu_s\perp m \). Mostramos por separado el comportamiento de la derivada simétrica de cada parte.

    \textbf{Paso 1. Parte absolutamente continua.}
    Por Radon-Nikodym existe \( f\in L_1(\R^k) \) tal que
    \[
        \mu_a(E)=\int_E f\,\mathrm dm
    \]
    para todo boreliano \( E \). Entonces
    \[
        \frac{\mu_a(B(x,r))}{m(B(x,r))}
        =
        \frac{1}{m(B(x,r))}\int_{B(x,r)} f\,\mathrm dm.
    \]
    Dado que las bolas \( B(x,r) \) se contraen adecuadamente a \( x \) (con \( \alpha=1 \) en la definición), el Teorema de Diferenciación de Lebesgue implica que este cociente converge a \( f(x) \) para todo punto de Lebesgue de \( f \). Como casi todo \( x\in\R^k \) es punto de Lebesgue de \( f \), se concluye que
    \[
        D\mu_a(x)=f(x)
        \qquad m\text{-c.t.p.}
    \]

    \textbf{Paso 2. Parte singular.}
    Basta tratar el caso \( \mu_s\ge 0 \); para una medida con signo o carga se reemplaza \( \mu_s \) por su variación total \( |\mu_s| \) y se usa que \( |D\mu_s(x)| \le D|\mu_s|(x) \).

    Como \( \mu_s\perp m \), existe un boreliano \( A \subset \R^k \) tal que
    \[
        m(A)=0,
        \qquad
        \mu_s(\R^k\setminus A)=0.
    \]
    Fijemos un compacto \( K\subset \R^k\setminus A \). Entonces \( \mu_s(K)=0 \). Sea \( \varepsilon>0 \). Por regularidad exterior de \( \mu_s \), existe un abierto \( V\supset K \) con \( \mu_s(V)<\varepsilon \). Definimos
    \[
        \mu_\varepsilon(E)=\mu_s(E\cap V).
    \]
    Si \( x\in K \), como \( V \) es abierto, existe \( r_0>0 \) tal que \( B(x,r)\subset V \) para todo \( r<r_0 \). Por lo tanto, para \( r<r_0 \):
    \[
        Q_r\mu_s(x)=Q_r\mu_\varepsilon(x).
    \]
    Tomando \( \limsup_{r\to 0} \), se obtiene
    \[
        D\mu_s(x) = \limsup_{r\to 0} Q_r\mu_s(x) = \limsup_{r\to 0} Q_r\mu_\varepsilon(x) \le M\mu_\varepsilon(x), \qquad x\in K.
    \]
    Para \( \lambda>0 \), definimos
    \[
        E_{\lambda,K}=\{x\in K: D\mu_s(x)>\lambda \}.
    \]
    Entonces \( E_{\lambda,K}\subset \{x\in K: M\mu_\varepsilon(x)>\lambda \} \), y por el Teorema Maximal,
    \[
        m(E_{\lambda,K}) \le \frac{3^k}{\lambda}\,\mu_\varepsilon(\R^k) = \frac{3^k}{\lambda}\,\mu_s(V) < \frac{3^k}{\lambda}\,\varepsilon.
    \]
    Como \( \varepsilon \) es arbitrario, \( m(E_{\lambda,K})=0 \). Es decir, para todo compacto \( K\subset\R^k\setminus A \),
    \[
        m\big( \{ x \in K: D\mu_s(x) > \lambda \} \big) = 0.
    \]

    Definimos ahora \( G_n=(\R^k\setminus A)\cap B(0,n) \) y \( \mathcal{G}_n = \{ x \in G_n : D\mu_s(x) > \lambda \} \). Para cada compacto \( K\subset \mathcal{G}_n \), se tiene \( K\subset \R^k\setminus A \) y todo punto de \( K \) satisface \( D\mu_s(x)>\lambda \), luego \( \{x\in K: D\mu_s(x)>\lambda\}=K \). Por lo ya demostrado, \( m(K)=0 \). Aplicando regularidad interior de \( m \) al conjunto medible \( \mathcal{G}_n \):
    \[
        m(\mathcal{G}_n)=\sup \{m(K): K\subset \mathcal{G}_n,\ K \text{ compacto}\}=0.
    \]
    Tomando unión en \( n \), se obtiene
    \[
        m\big( \{ x \in \R^k\setminus A: D\mu_s(x)> \lambda \} \big) = 0.
    \]
    Como \( m(A)=0 \), resulta
    \[
        m\big( \{x \in \R^k: D\mu_s(x) > \lambda \} \big)=0
        \quad\text{para todo }\lambda>0.
    \]
    Luego \( D\mu_s(x)=0 \) \( m \)-c.t.p.

    \textbf{Paso 3. Conclusión.}
    Para todo \( r>0 \) se cumple \( Q_r\mu(x)=Q_r\mu_a(x)+Q_r\mu_s(x) \). En el conjunto de medida plena donde existen los límites de ambos cocientes (garantizado por los pasos anteriores), tomando \( r\to 0 \):
    \[
        D\mu(x)=D\mu_a(x)+D\mu_s(x)=f(x)+0=f(x).
    \]
    Por lo tanto, \( D\mu=f \) \( m \)-c.t.p. Si además \( \mu\ll m \), entonces \( \mu_s=0 \) y
    \[
        \mu(E)=\int_E f\,\mathrm dm=\int_E D\mu \, \mathrm dm
    \]
    para todo boreliano \( E \).
\end{proof}

\section{Extras}

\begin{teorema}
    Si \( 1 \leq p < \infty \) y \( f \in L_p(\R^k) \), dado \( \varepsilon > 0 \)
    existe \( \varphi \in L_p \) simple tal que \( \| f - \varphi \|_p < \varepsilon \).
\end{teorema}
\begin{proof}
    Supongamos primero que \( f \geq 0 \). Por hipótesis \[
        \int f^p < \infty
    \]
    y que existe \( {(\varphi_n)}_{n \in \N} \) una sucesión de funciones simples, no negativas, monótonamente crecientes tal que \[
        f(x) = \lim \varphi_n(x)
    \]
    Como \( \varphi_n \leq f \), se cumple que \( \varphi_n^p \leq f^p \quad \forall n \in \N \) y entonces \( \int \varphi_n^p \leq \int f^p < \infty \implies \varphi_n \in L_p \quad \forall n \in \N \).
    Finalmente, como \( 0 \leq {(f - \varphi_n)}^p \leq f^p \) por TCD: \[
        \lim \int {(f - \varphi_n)}^p = \int \lim {(f - \varphi_n)}^p = 0
    \]
    O sea que \( \lim \| f - \varphi_n \|_p = 0 \), lo que implica que existe \( n_0 \in \N \) tal que \( \| f - \varphi_{n_0} \|_p < \varepsilon \).
    Para el caso donde \( f \) no es necesariamente no negativa, recordemos que \( f = f^+ - f^- \) con \( f^+, f^- \geq 0 \)
    Por lo tanto, elijo \( \| f^+ - \varphi_1 \| < \varepsilon / 2 \) y \( \| f^- - \varphi_2 \| < \varepsilon/2 \) y tomando \( \varphi = \varphi_1 - \varphi_2 \) vale.
\end{proof}

\begin{teorema}
    Análogo al teorema anterior, pero con \( \varphi \) continua en lugar de simple.
\end{teorema}
\begin{proof}
    El objetivo es aproximar a una función simple por funciones continuas. Si \( f = \chi_{E} \) con \( m(E) < \infty \) para que \( f \in L_p \).
    Dado \( \varepsilon > 0 \), existe \( F \) cerrado y \( G \) abierto tales que \( F \subseteq E \subseteq G \) y
    \( m(G \setminus F) < \varepsilon^p \).
    Defino: \[
        g(x) = \dfrac{d(x, G^c)}{d(x, F) + d(x, G^c)}
    \]
    Notar que el numerador se anula sii \( x \in G^c \). Como \( F \subseteq G \), la distancia a ambos conjuntos no puede ser simultáneamente cero, por lo que el denominador no se anula y \( 0 \leq g \leq 1 \) resulta en una función continua.
    Además, \( g(x) = 1 \) para \( x \in F \) y \( g(x) = 0 \) para \( x \in G^c \). Por lo tanto: \[
        \| \chi_E - g \|_p^p = \int_{\R^n} |\chi_E - g|^p = \int_{G \setminus F} |\chi_E - g|^p \leq m(G \setminus F) < \varepsilon^p
    \]
    Finalmente: \[
        \| \chi_E - g \|_p \leq \varepsilon
    \]
    Si \( \varphi \) es simple entonces: \[
        \varphi = \sum_{i=1}^n a_i \chi_{E_i} \implies |\varphi|^p = \sum_{i=1}^n |a_i|^p \chi_{E_i}
    \]
    Vale que: \[
        \| \varphi \|_p^p = \sum_{i=1}^n |a_i|^p m(E_i) < \infty
    \]
    Por lo anterior, para cada \( i \in \{ 1, \ldots, n \} \) existe \( g_i \in L_p \) continua tal que \( \| \chi_{E_i} - g_i \|_p < \varepsilon \)
    Sea \( g = \sum_{i=1}^n a_i g_i \), entonces \( g \) es continua y: \[
        \| \varphi - g \|_p = \left \| \sum_{i=1}^n a_i (\chi_{E_i} - g_i) \right \|_p \leq \sum_{i=1}^n |a_i| \| \chi_{E_i} - g_i \|_p < \varepsilon \sum_{i=1}^n |a_i|
    \]
    Por lo tanto, las funciones simples (que son densas) se pueden aproximar por funciones continuas.
\end{proof}

\begin{definicion}
    El soporte de una función \( f : \R^k \to \R \) es el conjunto: \[
        sop(f) = \overline{\{ x \in \R^k : f(x) \neq 0 \}}
    \]
    y \( C_c(\R^k) \) es el conjunto de funciones continuas con soporte compacto.
\end{definicion}

\begin{teorema}
    El espacio \( C_c(\R^k) \) es denso en \( L_p(\R^k) \) para \( 1 \leq p < \infty \).
\end{teorema}
\begin{proof}
    Dado \( f \in L_p \) y \( \varepsilon > 0 \), existe \( g \in L_p \) continua tal que \( \| f - g \|_p < \varepsilon / 2 \).
    Quiero aproximar a \( g \) por una función continua de soporte compacto que también esté en \( L_p \). Defino: \[
        \psi_k(x) = \dfrac{d(x, {B(0, 2k)}^c)}{d(x, B(0, k)) + d(x, {B(0, 2k)}^c)}
    \]
    Análogo al caso anterior, \( \psi_k \) es continua, \( 0 \leq \psi_k \leq 1 \), \( \psi_k(x) = 1 \) para \( x \in B(0, k) \) y
    \( \psi_k(x) = 0 \) para \( x \in {B(0, 2k)}^c \). Por lo tanto, \( \psi_k \) tiene soporte compacto. Defino: \[
        g_k = \psi_k \cdot g
    \]
    Que tiene soporte compacto pues: \[
        \operatorname{sop}(g_k) \subseteq \operatorname{sop}(\psi_k) \cap \operatorname{sop}(g) \subseteq \overline{B}(0, 2k) \cap \operatorname{sop}(g)
    \]
    Veamos que las \( g_k \) aproximan a \( g \) en \( L_p \). Sabemos que: \[
        |g - g_k|^p = |g - \psi_k g|^p = |(1 - \psi_k) g|^p \leq |g|^p
    \]
    Como \( g \in L_p \) se tiene que \( |g|^p \in L_1 \), luego, por convergencia dominada: \[
        \lim \| g - g_k \|_p^p = \lim \int |g - g_k|^p = \int \lim |g - g_k|^p = \int \lim |(1 - \psi_k) g|^p = 0
    \]
    pues \( \lim \psi_k = 1 \). Luego, existe un \( k_0 \in \N \) tal que \( \| g - g_{k_0} \|_p < \varepsilon / 2 \). Finalmente, tomando \( h = g_{k_0} \) se tiene que \( h \in C_c(\R^k) \) y: \[
        \| f - h \|_p \leq \| f - g \|_p + \| g - h \|_p < \varepsilon
    \]
\end{proof}
